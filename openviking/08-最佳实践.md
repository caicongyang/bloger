## 1. 性能优化

### 1.1 异步处理

`OpenViking` 是同步类，**异步 API 必须用 `AsyncOpenViking`**：

```python
import asyncio
import openviking as ov

async def main():
    client = ov.AsyncOpenViking(path="./data")   # ⚠️ 不是 ov.OpenViking
    await client.initialize()

    # 并发添加多个资源
    await asyncio.gather(
        client.add_resource(path="./docs1"),
        client.add_resource(path="./docs2"),
        client.add_resource(path="./docs3"),
    )
    await client.wait_processed()

    # 并发搜索
    results = await asyncio.gather(
        client.find("认证", target_uri="viking://resources/auth"),
        client.find("API",   target_uri="viking://resources/api"),
        client.find("配置",  target_uri="viking://resources/config"),
    )

    await client.close()

asyncio.run(main())
```

### 1.3 缓存策略

```mermaid
flowchart LR
    Query[查询] --> Cache{缓存?}
    Cache -->|Yes| Return[直接返回]
    Cache -->|No| Search[向量搜索]
    Search --> Store[存入缓存]
    Store --> Return
```

```python
# 使用 L0 缓存减少 LLM 调用
def get_content(uri, use_cache=True):
    # 先尝试 L0
    abstract = client.abstract(uri)
    
    if needs_more(abstract):
        overview = client.overview(uri)
        if needs_more(overview):
            return client.read(uri)
        return overview
    return abstract
```

## 3. 检索策略

### 3.1 渐进式加载

```python
def smart_retrieve(query, target_uri):
    # 1. L0 快速筛选
    results = client.find(query, target_uri=target_uri, limit=20)
    
    # 2. L1 精筛
    candidates = []
    for r in results.resources:
        overview = client.overview(r.uri)
        if relevant(overview, query):
            candidates.append(r)
    
    # 3. 按需加载 L2
    final = []
    for c in candidates[:5]:
        if needs_detail(c):
            content = client.read(c.uri)
            final.append({"uri": c.uri, "content": content, "score": c.score})
        else:
            final.append({"uri": c.uri, "overview": client.overview(c.uri), "score": c.score})
    
    return final
```

### 3.2 意图预判

```python
# 根据查询类型选择检索策略
def select_strategy(query):
    if query.startswith("如何") or query.startswith("怎么"):
        return "howto"  # 需要详细步骤
    elif query.startswith("什么是"):
        return "definition"  # 需要定义解释
    elif any(kw in query for kw in ["配置", "设置", "安装"]):
        return "guide"  # 需要指南
    else:
        return "general"
```

### 3.3 多路召回

```python
def multi_retrieve(query, target_uri):
    # 1. 语义搜索
    semantic_results = client.find(query, target_uri=target_uri)
    
    # 2. 关键词搜索
    keyword_results = client.grep(keyword, uri=target_uri)
    
    # 3. 关联搜索
    relations = client.relations(target_uri)
    
    # 4. 合并去重
    merged = merge_results(
        semantic_results,
        keyword_results,
        relations
    )
    
    return merged
```

## 4. 成本控制

### 4.1 Token 优化

| 策略 | 节省比例 | 适用场景 |
|------|----------|----------|
| L0 预筛选 | 60-80% | 大规模检索 |
| L1 导航 | 40-60% | 需要概览 |
| 按需 L2 | 30-50% | 详细阅读 |

```python
# 预算感知的检索
def budget_retrieve(query, target_uri, token_budget=2000):
    # 估算 token 使用
    results = client.find(query, target_uri=target_uri)
    
    total_tokens = 0
    selected = []
    
    for r in results.resources:
        # 加载 L1
        overview = client.overview(r.uri)
        tokens = estimate_tokens(overview)
        
        if total_tokens + tokens <= token_budget:
            selected.append({"uri": r.uri, "content": overview})
            total_tokens += tokens
        else:
            # 不足时用 L0
            selected.append({"uri": r.uri, "content": r.abstract})
            total_tokens += 100
    
    return selected
```

## 5. 调试技巧

### 5.1 查看检索轨迹

`search()` 返回的 `FindResult.query_plan` 包含意图分析的真实结构（`QueryPlan` 来自 `openviking_cli/retrieve/types.py`）：

```python
session = client.session()
session.add_message("user", [TextPart(text="帮我写一个登录接口")])

results = client.search("帮我写一个登录接口", session=session)

if results.query_plan:
    print("Reasoning:", results.query_plan.reasoning)
    print("Session context:", results.query_plan.session_context)
    for q in results.query_plan.queries:
        print(f"  - type={q.context_type}, priority={q.priority}, target_dirs={q.target_directories}")
        print(f"    query={q.query!r}")

# 每个 QueryResult 还包含 thinking_trace（用于可视化检索路径）
if results.query_results:
    for qr in results.query_results:
        print(f"searched_directories={qr.searched_directories}")
        print(f"thinking_trace={qr.thinking_trace}")
```

> ⚠️ `OpenViking` 构造函数**没有 `debug=True` 参数**；启用 `telemetry` 请在 API 调用时传 `telemetry=True`，例如 `client.find(query, telemetry=True)`。

### 5.2 查看系统状态

```python
status = client.get_status()    # SystemStatus（各组件健康状态）
print(client.is_healthy())

# 通过 observer 获取组件级状态
observer = client.observer
```

> ⚠️ **不存在 `client.debug_vector(uri)`**。如要观察具体 URI 的向量数据，请用 HTTP API `/api/v1/observer/...` 或 `/api/v1/debug/...`（详见 `openviking/server/routers/observer.py`、`debug.py`）。

### 5.3 检查会话状态

```python

def build_agent_context(task: str):
    # 1. Agent 技能
    skills = client.find(task, target_uri="viking://agent/skills/")

    memory = client.find(task, target_uri="viking://agent/memories/")
```

### 6.2 RAG 管道

```python
def rag_pipeline(query, documents):
    # 1. 添加文档（建议 wait=True，避免后续读 L1 时未生成）
    for doc in documents:
        client.add_resource(path=doc, wait=True)

    # 2. 检索
    results = client.find(query, target_uri="viking://resources/")

    # 3. 构建 Prompt（使用 L1 概览）
    context_pieces = []
    for r in results.resources[:3]:
        try:
            ov_text = client.overview(r.uri)
        except Exception:
            ov_text = r.abstract
        context_pieces.append(f"## {r.uri}\n{ov_text}")
    context = "\n\n".join(context_pieces)

    prompt = f"""基于以下上下文回答问题:

{context}

问题: {query}

回答:"""
    return prompt
```

### 6.3 用户隔离

OpenViking 内置 user/agent scope；要做"租户隔离"，建议使用 `viking://user/<user_id>/...` 这一层而非自造前缀：

```python
class TenantClient:
    def __init__(self, client, user_id: str):
        self.client = client
        self.user_id = user_id

    @property
    def base_uri(self) -> str:
        # user 顶级 scope，OpenViking 标准路径
        return f"viking://user/{self.user_id}/"

    def find(self, query, **kwargs):
        target = kwargs.pop("target_uri", self.base_uri)
        return self.client.find(query, target_uri=target, **kwargs)

    def add_resource(self, path, **kwargs):
        return self.client.add_resource(
            path,
            parent=f"{self.base_uri}resources/",
            reason=f"user:{self.user_id}",
            **kwargs,
        )
```

> 多租户的强隔离推荐使用 HTTP Server 模式 + `account_id` / `owner_user_id`（参见 `ContextCollection` schema），由服务端统一鉴权。
